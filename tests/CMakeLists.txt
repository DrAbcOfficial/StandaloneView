cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(StandaloneViewTests LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "StandaloneView tests require Windows MSVC x86 (-A Win32).")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(METAHOOK_SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../MetaHook" CACHE PATH "MetaHook SDK source tree")
enable_testing()
include(../cmake/SDK.cmake)
add_executable(standaloneview_manager_test manager_test.cpp ../src/StandaloneViewManager.cpp ../src/EmptyWorld.cpp)
standaloneview_use_sdk(standaloneview_manager_test)
add_executable(standaloneview_hooks_test hooks_test.cpp ../src/privatehook.cpp ../src/StandaloneViewManager.cpp ../src/EmptyWorld.cpp)
standaloneview_use_sdk(standaloneview_hooks_test)
add_test(NAME standaloneview_manager COMMAND standaloneview_manager_test)
add_test(NAME standaloneview_hooks COMMAND standaloneview_hooks_test)
set_tests_properties(standaloneview_manager standaloneview_hooks PROPERTIES TIMEOUT 30)
